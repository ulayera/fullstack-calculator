package expression

import "fmt"

type Kind int

const (
        Number Kind = iota
        Plus
        Minus
        Star
        Slash
        LParen
        RParen
)

type Token struct {
        Kind Kind
        Text string
        Pos  int
}

var symbols = map[rune]Kind{
        '+': Plus,
        '-': Minus,
        '*': Star,
        '/': Slash,
        '(': LParen,
        ')': RParen,
}

func isNumberRune(r rune) bool {
        return r == '.' || (r >= '0' && r <= '9')
}

func Lex(input string) ([]Token, []string) {
        runes := []rune(input)
        var tokens []Token
        var errs []string
        for i := 0; i < len(runes); i++ {
                r := runes[i]
                if kind, ok := symbols[r]; ok {
                        tokens = append(tokens, Token{Kind: kind, Text: string(r), Pos: i + 1})
                        continue
                }
                if !isNumberRune(r) {
                        errs = append(errs, fmt.Sprintf("Invalid character '%c' at position %d", r, i+1))
                        continue
                }
                start := i
                for i+1 < len(runes) && isNumberRune(runes[i+1]) {
                        i++
                }
                tokens = append(tokens, Token{Kind: Number, Text: string(runes[start : i+1]), Pos: start + 1})
        }
        return tokens, errs
}

import (
        "errors"

        "github.com/shopspring/decimal"
)

const DivisionPrecision = 34

var ErrDivisionByZero = errors.New("division by zero")

type Node interface {
        Eval() (decimal.Decimal, error)
}

type Literal struct {
        Value decimal.Decimal
}

func (l Literal) Eval() (decimal.Decimal, error) {
        return l.Value, nil
}

type Negate struct {
        Operand Node
}

func (n Negate) Eval() (decimal.Decimal, error) {
        v, err := n.Operand.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        return v.Neg(), nil
}

type Binary struct {
        Op    Kind
        Left  Node
        Right Node
}

func (b Binary) Eval() (decimal.Decimal, error) {
        left, err := b.Left.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        right, err := b.Right.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        switch b.Op {
        case Plus:
                return left.Add(right), nil
        case Minus:
                return left.Sub(right), nil
        case Star:
                return left.Mul(right), nil
        }
        if right.IsZero() {
                return decimal.Decimal{}, ErrDivisionByZero
        }
        return left.DivRound(right, DivisionPrecision), nil
}

package expression

import "github.com/shopspring/decimal"

type descent struct {
        tokens []Token
        pos    int
}

type Parser struct{}

func (Parser) Parse(input string) (Node, []string) {
        tokens, errs := validate(input)
        if len(errs) > 0 {
                return nil, errs
        }
        p := &descent{tokens: tokens}
        return p.expression(), nil
}

func (p *descent) accept(kinds ...Kind) (Kind, bool) {
        if p.pos >= len(p.tokens) {
                return 0, false
        }
        k := p.tokens[p.pos].Kind
        for _, want := range kinds {
                if k == want {
                        p.pos++
                        return k, true
                }
        }
        return 0, false
}

func (p *descent) expression() Node {
        left := p.term()
        for {
                op, ok := p.accept(Plus, Minus)
                if !ok {
                        return left
                }
                left = Binary{Op: op, Left: left, Right: p.term()}
        }
}

func (p *descent) term() Node {
        left := p.unary()
        for {
                op, ok := p.accept(Star, Slash)
                if !ok {
                        return left
                }
                left = Binary{Op: op, Left: left, Right: p.unary()}
        }
}

func (p *descent) unary() Node {
        if _, ok := p.accept(Minus); ok {
                return Negate{Operand: p.unary()}
        }
        return p.primary()
}

func (p *descent) primary() Node {
        if _, ok := p.accept(LParen); ok {
                inner := p.expression()
                p.pos++
                return inner
        }
        t := p.tokens[p.pos]
        p.pos++
        return Literal{Value: decimal.RequireFromString(t.Text)}
}

package expression

import (
        "fmt"
        "strings"
        "unicode/utf8"
)

const MaxLength = 16384

func validate(input string) ([]Token, []string) {
        if input == "" {
                return nil, []string{"Expression is empty"}
        }
        if utf8.RuneCountInString(input) > MaxLength {
                return nil, []string{fmt.Sprintf("Expression exceeds %d characters", MaxLength)}
        }
        tokens, errs := Lex(input)
        if len(errs) > 0 {
                return nil, errs
        }
        errs = append(errs, checkNumbers(tokens)...)
        errs = append(errs, checkSequence(tokens)...)
        errs = append(errs, checkParentheses(tokens)...)
        return tokens, errs
}

func isOperator(k Kind) bool {
        return k == Plus || k == Minus || k == Star || k == Slash
}

func expectsOperand(prev *Token) bool {
        return prev == nil || prev.Kind == LParen || isOperator(prev.Kind)
}

func checkNumbers(tokens []Token) []string {
        var errs []string
        for _, t := range tokens {
                if t.Kind != Number {
                        continue
                }
                switch {
                case strings.Count(t.Text, ".") > 1:
                        errs = append(errs, fmt.Sprintf("Number '%s' at position %d has more than one decimal point", t.Text, t.Pos))
                case strings.HasSuffix(t.Text, "."):
                        errs = append(errs, fmt.Sprintf("Number '%s' at position %d cannot end with a decimal point", t.Text, t.Pos))
                }
        }
        return errs
}

func checkSequence(tokens []Token) []string {
        var errs []string
        var prev *Token
        prevIsSign := false
        for i := range tokens {
                t := &tokens[i]
                isSign := false
                switch t.Kind {
                case Number, LParen:
                        if !expectsOperand(prev) {
                                errs = append(errs, fmt.Sprintf("Implicit multiplication is not allowed at position %d", t.Pos))
                        }
                case RParen:
                        if prev != nil && prev.Kind == LParen {
                                errs = append(errs, fmt.Sprintf("Empty parentheses at position %d", prev.Pos))
                        } else if prev != nil && isOperator(prev.Kind) {
                                errs = append(errs, fmt.Sprintf("Missing operand after '%s' at position %d", prev.Text, prev.Pos))
                        }
                case Minus:
                        isSign = expectsOperand(prev)
                        if isSign && prevIsSign {
                                errs = append(errs, fmt.Sprintf("Minus sign at position %d repeats a sign; only one is allowed", t.Pos))
                        }
                default:
                        if expectsOperand(prev) {
                                errs = append(errs, fmt.Sprintf("Operator '%s' at position %d is missing its left operand", t.Text, t.Pos))
                        }
                }
                prevIsSign = isSign
                prev = t
        }
        if prev != nil && isOperator(prev.Kind) {
                errs = append(errs, fmt.Sprintf("Incomplete expression: missing operand after '%s' at position %d", prev.Text, prev.Pos))
        }
        return errs
}

func checkParentheses(tokens []Token) []string {
        var errs []string
        var open []int
        for _, t := range tokens {
                switch t.Kind {
                case LParen:
                        open = append(open, t.Pos)
                case RParen:
                        if len(open) == 0 {
                                errs = append(errs, fmt.Sprintf("Unmatched ')' at position %d", t.Pos))
                                continue
                        }
                        open = open[:len(open)-1]
                }
        }
        for _, pos := range open {
                errs = append(errs, fmt.Sprintf("Unclosed '(' at position %d", pos))
        }
        return errs
}

import (
        "net/http"

        "github.com/gin-gonic/gin"
        "github.com/ulayera/fullstack-calculator/backend/internal/service"
)

const (
        MaxBodyBytes   = 64 << 10
        InvalidBodyMsg = "Request body must be a JSON object with a string 'expression'"
        Ok             = "Ok"
)

type expressionRequest struct {
        Expression *string `json:"expression" binding:"required"`
}

type errorsResponse struct {
        Errors []string `json:"errors"`
}

type CalculatorHandler struct {
        service service.CalculatorService
}

func NewCalculatorHandler(s service.CalculatorService) *CalculatorHandler {
        return &CalculatorHandler{service: s}
}

func bindExpression(c *gin.Context) (string, bool) {
        c.Request.Body = http.MaxBytesReader(c.Writer, c.Request.Body, MaxBodyBytes)
        var req expressionRequest
        if err := c.ShouldBindJSON(&req); err != nil {
                c.JSON(http.StatusBadRequest, errorsResponse{Errors: []string{InvalidBodyMsg}})
                return "", false
        }
        return *req.Expression, true
}

func (h *CalculatorHandler) Validate(c *gin.Context) {
        input, ok := bindExpression(c)
        if !ok {
                return
        }
        if errs := h.service.Validate(input); len(errs) > 0 {
                c.JSON(http.StatusBadRequest, errorsResponse{Errors: errs})
                return
        }
        c.JSON(http.StatusOK, gin.H{"result": Ok})
}

func (h *CalculatorHandler) Calculate(c *gin.Context) {
        input, ok := bindExpression(c)
        if !ok {
                return
        }
        calc, errs, err := h.service.Calculate(input)
        if err != nil {
                _ = c.Error(err)
                c.JSON(http.StatusInternalServerError, errorsResponse{Errors: []string{"Internal error"}})
                return
        }
        if len(errs) > 0 {
                c.JSON(http.StatusBadRequest, errorsResponse{Errors: errs})
                return
        }
        c.JSON(http.StatusOK, calc)
}

import (
        "context"
        "net"
        "net/http"
        "time"
)

func New(handler http.Handler) *http.Server {
        return &http.Server{
                Handler:           handler,
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }
}

func Run(ctx context.Context, srv *http.Server, ln net.Listener, shutdownTimeout time.Duration) error {
        errs := make(chan error, 1)
        go func() { errs <- srv.Serve(ln) }()

        select {
        case err := <-errs:
                return err
        case <-ctx.Done():
                shutdownCtx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
                defer cancel()
                return srv.Shutdown(shutdownCtx)
        }
}

import (
        "errors"

        "github.com/ulayera/fullstack-calculator/backend/internal/expression"
)

const (
        ResultPlaces = 16
        Undefined    = "undefined"
)

type Parser interface {
        Parse(input string) (expression.Node, []string)
}

type Calculation struct {
        Result string `json:"result"`
        Exact  string `json:"exact,omitempty"`
}

type CalculatorService interface {
        Validate(input string) []string
        Calculate(input string) (Calculation, []string, error)
}

type calculatorService struct {
        parser Parser
}

func NewCalculatorService(p Parser) CalculatorService {
        return calculatorService{parser: p}
}

func (s calculatorService) Validate(input string) []string {
        _, errs := s.parser.Parse(input)
        return errs
}

func (s calculatorService) Calculate(input string) (Calculation, []string, error) {
        node, errs := s.parser.Parse(input)
        if len(errs) > 0 {
                return Calculation{}, errs, nil
        }
        value, err := node.Eval()
        if errors.Is(err, expression.ErrDivisionByZero) {
                return Calculation{Result: Undefined}, nil, nil
        }
        if err != nil {
                return Calculation{}, nil, err
        }
        return Calculation{Result: value.Round(ResultPlaces).String(), Exact: value.String()}, nil, nil
}
