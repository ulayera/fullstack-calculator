const PowerLimitVar = "POWER_LIMIT"

type Lookup func(key string) (string, bool)

func PowerLimit(lookup Lookup) (int64, error) {
        raw, ok := lookup(PowerLimitVar)
        if !ok || raw == "" {
                return expression.DefaultPowerLimit, nil
        }
        limit, err := strconv.ParseInt(raw, 10, 64)
        if err != nil || limit <= 0 || limit > expression.MaxPowerLimit {
                return 0, fmt.Errorf("%s must be an integer from 1 to %d, got %q", PowerLimitVar, expression.MaxPowerLimit, raw)
        }
        return limit, nil
}

package expression

import "fmt"

type Kind int

const (
        Number Kind = iota
        Plus
        Minus
        Star
        Slash
        LParen
        RParen
        Caret
        Percent
        Root
)

type Token struct {
        Kind Kind
        Text string
        Pos  int
}

var symbols = map[rune]Kind{
        '+': Plus,
        '-': Minus,
        '*': Star,
        '/': Slash,
        '(': LParen,
        ')': RParen,
        '^': Caret,
        '%': Percent,
        '√': Root,
        'V': Root,
        'v': Root,
}

func isNumberRune(r rune) bool {
        return r == '.' || (r >= '0' && r <= '9')
}

func Lex(input string) ([]Token, []string) {
        runes := []rune(input)
        var tokens []Token
        var errs []string
        for i := 0; i < len(runes); i++ {
                r := runes[i]
                if kind, ok := symbols[r]; ok {
                        tokens = append(tokens, Token{Kind: kind, Text: string(r), Pos: i + 1})
                        continue
                }
                if !isNumberRune(r) {
                        errs = append(errs, fmt.Sprintf("Invalid character '%c' at position %d", r, i+1))
                        continue
                }
                start := i
                for i+1 < len(runes) && isNumberRune(runes[i+1]) {
                        i++
                }
                tokens = append(tokens, Token{Kind: Number, Text: string(runes[start : i+1]), Pos: start + 1})
        }
        return tokens, errs
}

package expression

import (
        "errors"

        "github.com/shopspring/decimal"
)

const DivisionPrecision = 34

var (
        ErrDivisionByZero = errors.New("division by zero")
        one               = decimal.NewFromInt(1)
)

type Unprocessable string

func (u Unprocessable) Error() string {
        return string(u)
}

const (
        ErrImaginary  Unprocessable = "imaginary numbers not supported"
        ErrPowerLimit Unprocessable = "power operation refused due to computation limit exceeded"
)

type Node interface {
        Eval() (decimal.Decimal, error)
}

type Literal struct {
        Value decimal.Decimal
}

func (l Literal) Eval() (decimal.Decimal, error) {
        return l.Value, nil
}

type Negate struct {
        Operand Node
}

func (n Negate) Eval() (decimal.Decimal, error) {
        v, err := n.Operand.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        return v.Neg(), nil
}

type Binary struct {
        Op    Kind
        Left  Node
        Right Node
        Limit int64
}

func (b Binary) Eval() (decimal.Decimal, error) {
        left, err := b.Left.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        right, err := b.Right.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        v, err := b.apply(left, right)
        return bounded(v, err, b.Limit)
}

func (b Binary) apply(left, right decimal.Decimal) (decimal.Decimal, error) {
        switch b.Op {
        case Plus:
                return left.Add(right), nil
        case Minus:
                return left.Sub(right), nil
        case Star:
                return left.Mul(right), nil
        case Percent:
                return left.Mul(right).Shift(-2), nil
        }
        return divide(left, right)
}

func bounded(v decimal.Decimal, err error, limit int64) (decimal.Decimal, error) {
        if err != nil || v.IsZero() {
                return v, err
        }
        budget := max(limit, MaxLength) + DivisionPrecision
        lead := int64(leadingExponent(v))
        if int64(v.NumDigits()) > budget || lead > budget || -lead > budget {
                return decimal.Decimal{}, ErrPowerLimit
        }
        return v, nil
}

type SquareRoot struct {
        Operand Node
}

func (s SquareRoot) Eval() (decimal.Decimal, error) {
        v, err := s.Operand.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        if v.IsNegative() {
                return decimal.Decimal{}, ErrImaginary
        }
        if v.IsZero() {
                return v, nil
        }
        return v.SqrtRound(DivisionPrecision - 1 - leadingExponent(v)>>1)
}

func divide(left, right decimal.Decimal) (decimal.Decimal, error) {
        if right.IsZero() {
                return decimal.Decimal{}, ErrDivisionByZero
        }
        return left.DivRound(right, divisionPlaces(left, right)), nil
}

func leadingExponent(d decimal.Decimal) int32 {
        return int32(d.NumDigits()) + d.Exponent() - 1
}

func divisionPlaces(left, right decimal.Decimal) int32 {
        if left.IsZero() {
                return 0
        }
        exponent := leadingExponent(left) - leadingExponent(right)
        if right.Abs().Shift(exponent).Cmp(left.Abs()) > 0 {
                exponent--
        }
        return DivisionPrecision - 1 - exponent
}

package expression

import "github.com/shopspring/decimal"

type descent struct {
        tokens     []Token
        pos        int
        powerLimit int64
        work       *Work
}

type Parser struct {
        PowerLimit int64
}

func (parser Parser) Parse(input string) (Node, []string) {
        tokens, errs := validate(input)
        if len(errs) > 0 {
                return nil, errs
        }
        p := &descent{tokens: tokens, powerLimit: parser.PowerLimit, work: NewWork(parser.PowerLimit)}
        return p.expression(), nil
}

func (p *descent) accept(kinds ...Kind) (Kind, bool) {
        if p.pos >= len(p.tokens) {
                return 0, false
        }
        k := p.tokens[p.pos].Kind
        for _, want := range kinds {
                if k == want {
                        p.pos++
                        return k, true
                }
        }
        return 0, false
}

func (p *descent) expression() Node {
        left := p.term()
        for {
                op, ok := p.accept(Plus, Minus)
                if !ok {
                        return left
                }
                left = Binary{Op: op, Left: left, Right: p.term(), Limit: p.powerLimit}
        }
}

func (p *descent) term() Node {
        left := p.unary()
        for {
                op, ok := p.accept(Star, Slash, Percent)
                if !ok {
                        return left
                }
                left = Binary{Op: op, Left: left, Right: p.unary(), Limit: p.powerLimit}
        }
}

func (p *descent) unary() Node {
        if _, ok := p.accept(Minus); ok {
                return Negate{Operand: p.unary()}
        }
        if _, ok := p.accept(Root); ok {
                return SquareRoot{Operand: p.unary()}
        }
        return p.power()
}

func (p *descent) power() Node {
        base := p.primary()
        if _, ok := p.accept(Caret); ok {
                return Power{Base: base, Exponent: p.unary(), Limit: p.powerLimit, Work: p.work}
        }
        return base
}

func (p *descent) primary() Node {
        if _, ok := p.accept(LParen); ok {
                inner := p.expression()
                p.pos++
                return inner
        }
        t := p.tokens[p.pos]
        p.pos++
        return Literal{Value: decimal.RequireFromString(t.Text)}
}

package expression

import (
        "math"
        "strconv"

        "github.com/shopspring/decimal"
)

const (
        DefaultPowerLimit = 100000
        MaxPowerLimit     = 100000000
        mantissaDigits    = 17
        guardDigits       = 10
        workFactor        = 10
)

type Work struct {
        remaining int64
}

func NewWork(limit int64) *Work {
        return &Work{remaining: workFactor * limit}
}

func (w *Work) spend(digits int64) bool {
        w.remaining -= digits
        return w.remaining >= 0
}

var nearOne = decimal.New(1, -2)

type Power struct {
        Base     Node
        Exponent Node
        Limit    int64
        Work     *Work
}

func (p Power) Eval() (decimal.Decimal, error) {
        base, err := p.Base.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        exponent, err := p.Exponent.Eval()
        if err != nil {
                return decimal.Decimal{}, err
        }
        v, err := power(base, exponent, p.Limit, p.Work)
        return bounded(v, err, p.Limit)
}

func power(base, exponent decimal.Decimal, limit int64, work *Work) (decimal.Decimal, error) {
        switch {
        case exponent.IsZero():
                return one, nil
        case base.IsZero() && exponent.IsNegative():
                return decimal.Decimal{}, ErrDivisionByZero
        case base.IsZero():
                return base, nil
        case base.IsNegative() && !exponent.IsInteger():
                return decimal.Decimal{}, ErrImaginary
        }
        magnitude := log10Power(base, exponent)
        if math.Abs(magnitude) > float64(limit) {
                return decimal.Decimal{}, ErrPowerLimit
        }
        if !exponent.IsInteger() || exponent.IsNegative() {
                v, err := base.PowWithPrecision(exponent, significantPlaces(magnitude))
                if err != nil {
                        return decimal.Decimal{}, ErrPowerLimit
                }
                return v.Round(DivisionPrecision - 1 - leadingExponent(v)), nil
        }
        if digits := exactDigits(base, exponent, magnitude); digits.LessThanOrEqual(decimal.NewFromInt(limit + 1)) {
                if !work.spend(digits.IntPart()) {
                        return decimal.Decimal{}, ErrPowerLimit
                }
                return base.PowBigInt(exponent.BigInt())
        }
        inverse, err := base.PowWithPrecision(exponent.Neg(), significantPlaces(-magnitude))
        if err != nil {
                return decimal.Decimal{}, ErrPowerLimit
        }
        return divide(one, inverse)
}

func significantPlaces(magnitude float64) int32 {
        return DivisionPrecision - 1 + guardDigits - int32(math.Floor(magnitude))
}

func exactDigits(base, exponent decimal.Decimal, magnitude float64) decimal.Decimal {
        whole := decimal.NewFromFloat(math.Floor(max(magnitude, 0)) + 1)
        scale := decimal.NewFromInt(int64(max(-base.Exponent(), 0)))
        return whole.Add(exponent.Mul(scale))
}

func log10Power(base, exponent decimal.Decimal) float64 {
        sign := float64(exponent.Sign())
        if base.Abs().LessThan(one) {
                sign = -sign
        }
        return sign * math.Pow(10, log10Abs(exponent)+log10Log10Abs(base))
}

func log10Log10Abs(d decimal.Decimal) float64 {
        abs := d.Abs()
        distance := abs.Sub(one)
        if distance.IsZero() {
                return math.Inf(-1)
        }
        if distance.Abs().GreaterThanOrEqual(nearOne) {
                return math.Log10(math.Abs(log10Abs(abs)))
        }
        ratio := 1.0
        if t := distance.InexactFloat64(); t != 0 {
                ratio = math.Log1p(t) / t
        }
        return log10Abs(distance) + math.Log10(ratio/math.Ln10)
}

func log10Abs(d decimal.Decimal) float64 {
        digits := d.Coefficient()
        digits.Abs(digits)
        text := digits.String()
        lead := float64(len(text)-1) + float64(d.Exponent())
        mantissa, _ := strconv.ParseFloat(text[:1]+"."+text[1:min(len(text), mantissaDigits)], 64)
        return lead + math.Log10(mantissa)
}

package expression

import (
        "fmt"
        "strings"
        "unicode/utf8"
)

const MaxLength = 16384

func validate(input string) ([]Token, []string) {
        if input == "" {
                return nil, []string{"Expression is empty"}
        }
        if utf8.RuneCountInString(input) > MaxLength {
                return nil, []string{fmt.Sprintf("Expression exceeds %d characters", MaxLength)}
        }
        tokens, errs := Lex(input)
        if len(errs) > 0 {
                return nil, errs
        }
        errs = append(errs, checkNumbers(tokens)...)
        errs = append(errs, checkParentheses(tokens)...)
        errs = append(errs, checkSequence(tokens)...)
        return tokens, errs
}

func isBinary(k Kind) bool {
        return k == Plus || k == Minus || k == Star || k == Slash || k == Caret || k == Percent
}

func needsOperand(k Kind) bool {
        return isBinary(k) || k == Root
}

func expectsOperand(prev *Token) bool {
        return prev == nil || prev.Kind == LParen || needsOperand(prev.Kind)
}

func checkNumbers(tokens []Token) []string {
        var errs []string
        for _, t := range tokens {
                if t.Kind != Number {
                        continue
                }
                switch {
                case strings.Count(t.Text, ".") > 1:
                        errs = append(errs, fmt.Sprintf("Number '%s' at position %d has more than one decimal point", t.Text, t.Pos))
                case strings.HasSuffix(t.Text, "."):
                        errs = append(errs, fmt.Sprintf("Number '%s' at position %d cannot end with a decimal point", t.Text, t.Pos))
                }
        }
        return errs
}

func checkSequence(tokens []Token) []string {
        var errs []string
        var prev *Token
        prevIsSign := false
        for i := range tokens {
                t := &tokens[i]
                isSign := false
                switch t.Kind {
                case Number, LParen, Root:
                        if !expectsOperand(prev) {
                                errs = append(errs, fmt.Sprintf("Implicit multiplication is not allowed at position %d", t.Pos))
                        }
                case RParen:
                        if prev != nil && prev.Kind == LParen {
                                errs = append(errs, fmt.Sprintf("Empty parentheses at position %d", prev.Pos))
                        } else if prev != nil && needsOperand(prev.Kind) {
                                errs = append(errs, fmt.Sprintf("Missing operand after '%s' at position %d", prev.Text, prev.Pos))
                        }
                case Minus:
                        isSign = expectsOperand(prev)
                        if isSign && prevIsSign {
                                errs = append(errs, fmt.Sprintf("Minus sign at position %d repeats a sign; only one is allowed", t.Pos))
                        }
                default:
                        if expectsOperand(prev) {
                                errs = append(errs, fmt.Sprintf("Operator '%s' at position %d is missing its left operand", t.Text, t.Pos))
                        }
                }
                prevIsSign = isSign
                prev = t
        }
        if prev != nil && needsOperand(prev.Kind) {
                errs = append(errs, fmt.Sprintf("Incomplete expression: missing operand after '%s' at position %d", prev.Text, prev.Pos))
        }
        return errs
}

func checkParentheses(tokens []Token) []string {
        var errs []string
        var open []int
        for _, t := range tokens {
                switch t.Kind {
                case LParen:
                        open = append(open, t.Pos)
                case RParen:
                        if len(open) == 0 {
                                errs = append(errs, fmt.Sprintf("Unmatched ')' at position %d", t.Pos))
                                continue
                        }
                        open = open[:len(open)-1]
                }
        }
        for _, pos := range open {
                errs = append(errs, fmt.Sprintf("Unclosed '(' at position %d", pos))
        }
        return errs
}

import (
        "errors"
        "net/http"

        "github.com/gin-gonic/gin"
        "github.com/ulayera/fullstack-calculator/backend/internal/expression"
        "github.com/ulayera/fullstack-calculator/backend/internal/service"
)

const (
        MaxBodyBytes   = 64 << 10
        InvalidBodyMsg = "Request body must be a JSON object with a string 'expression'"
        InternalMsg    = "Internal error"
        Ok             = "Ok"
)

type expressionRequest struct {
        Expression *string `json:"expression" binding:"required"`
}

type errorsResponse struct {
        Errors []string `json:"errors"`
}

type CalculatorHandler struct {
        service service.CalculatorService
}

func NewCalculatorHandler(s service.CalculatorService) *CalculatorHandler {
        return &CalculatorHandler{service: s}
}

func bindExpression(c *gin.Context) (string, bool) {
        c.Request.Body = http.MaxBytesReader(c.Writer, c.Request.Body, MaxBodyBytes)
        var req expressionRequest
        if err := c.ShouldBindJSON(&req); err != nil {
                c.JSON(http.StatusBadRequest, errorsResponse{Errors: []string{InvalidBodyMsg}})
                return "", false
        }
        return *req.Expression, true
}

func respondFailure(c *gin.Context, errs []string, err error) bool {
        var unprocessable expression.Unprocessable
        switch {
        case errors.As(err, &unprocessable):
                c.JSON(http.StatusUnprocessableEntity, errorsResponse{Errors: []string{unprocessable.Error()}})
        case err != nil:
                _ = c.Error(err)
                c.JSON(http.StatusInternalServerError, errorsResponse{Errors: []string{InternalMsg}})
        case len(errs) > 0:
                c.JSON(http.StatusBadRequest, errorsResponse{Errors: errs})
        default:
                return false
        }
        return true
}

func (h *CalculatorHandler) Validate(c *gin.Context) {
        input, ok := bindExpression(c)
        if !ok {
                return
        }
        if errs, err := h.service.Validate(input); respondFailure(c, errs, err) {
                return
        }
        c.JSON(http.StatusOK, gin.H{"result": Ok})
}

func (h *CalculatorHandler) Calculate(c *gin.Context) {
        input, ok := bindExpression(c)
        if !ok {
                return
        }
        calc, errs, err := h.service.Calculate(input)
        if respondFailure(c, errs, err) {
                return
        }
        c.JSON(http.StatusOK, calc)
}

import (
        "context"

        "net/http"
        "time"
)

func New(handler http.Handler) *http.Server {
        return &http.Server{
                Handler:           handler,
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }
}

func Run(ctx context.Context, srv *http.Server, ln net.Listener, shutdownTimeout time.Duration) error {
        errs := make(chan error, 1)
        go func() { errs <- srv.Serve(ln) }()

        select {
        case err := <-errs:
                return err
        case <-ctx.Done():
                shutdownCtx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
                defer cancel()
                return srv.Shutdown(shutdownCtx)
        }
}

package service

import (
        "errors"

        "github.com/shopspring/decimal"
        "github.com/ulayera/fullstack-calculator/backend/internal/expression"
)

const (
        ResultPlaces = 16
        Undefined    = "undefined"
)

type Parser interface {
        Parse(input string) (expression.Node, []string)
}

type Calculation struct {
        Result string `json:"result"`
        Exact  string `json:"exact,omitempty"`
}

type CalculatorService interface {
        Validate(input string) ([]string, error)
        Calculate(input string) (Calculation, []string, error)
}

type calculatorService struct {
        parser Parser
}

func NewCalculatorService(p Parser) CalculatorService {
        return calculatorService{parser: p}
}

func (s calculatorService) Validate(input string) ([]string, error) {
        _, _, errs, err := s.evaluate(input)
        return errs, err
}

func (s calculatorService) Calculate(input string) (Calculation, []string, error) {
        value, undefined, errs, err := s.evaluate(input)
        switch {
        case len(errs) > 0 || err != nil:
                return Calculation{}, errs, err
        case undefined:
                return Calculation{Result: Undefined}, nil, nil
        }
        return Calculation{Result: value.Round(ResultPlaces).String(), Exact: value.String()}, nil, nil
}

func (s calculatorService) evaluate(input string) (decimal.Decimal, bool, []string, error) {
        node, errs := s.parser.Parse(input)
        if len(errs) > 0 {
                return decimal.Decimal{}, false, errs, nil
        }
        value, err := node.Eval()
        if errors.Is(err, expression.ErrDivisionByZero) {
                return decimal.Decimal{}, true, nil, nil
        }
        return value, false, nil, err
}
